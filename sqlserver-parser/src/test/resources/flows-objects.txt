===OBJECT:fin.fn_AccountBalance:SQL_SCALAR_FUNCTION===-- scalar: account balance (correlated subquery) — function → function via fn_ConvertCurrency
CREATE FUNCTION fin.fn_AccountBalance(@accountId INT, @asOf DATE)
RETURNS DECIMAL(18,2)
AS
BEGIN
    DECLARE @bal DECIMAL(18,2), @ccy NVARCHAR(3);
    SELECT @ccy = Currency FROM fin.Account WHERE AccountId = @accountId;
    SELECT @bal = SUM(jl.Debit - jl.Credit)
    FROM fin.JournalLine jl
    INNER JOIN fin.JournalEntry je ON je.EntryId = jl.EntryId
    WHERE jl.AccountId = @accountId AND je.EntryDate <= @asOf;
    RETURN fin.fn_ConvertCurrency(ISNULL(@bal,0), @ccy, @asOf);   -- function → function
END;
===END===
===OBJECT:fin.fn_AccountTree:SQL_TABLE_VALUED_FUNCTION===-- multi-statement TVF: recursive account hierarchy (roll-up tree)
CREATE FUNCTION fin.fn_AccountTree(@rootId INT)
RETURNS @tree TABLE(Level INT, AccountId INT, ParentAccountId INT, Code NVARCHAR(20))
AS
BEGIN
    WITH acct AS (
        SELECT 0 AS Lvl, a.AccountId, a.ParentAccountId, a.Code
        FROM fin.Account a WHERE a.AccountId = @rootId
        UNION ALL
        SELECT p.Lvl + 1, a.AccountId, a.ParentAccountId, a.Code
        FROM fin.Account a
        INNER JOIN acct p ON a.ParentAccountId = p.AccountId
    )
    INSERT INTO @tree SELECT Lvl, AccountId, ParentAccountId, Code FROM acct;
    RETURN;
END;
===END===

CREATE FUNCTION fin.fn_ConvertCurrency(@amount DECIMAL(18,2), @ccy NVARCHAR(3), @asOf DATE)
RETURNS DECIMAL(18,2)
AS
BEGIN
    DECLARE @rate DECIMAL(18,6);
    SELECT TOP 1 @rate = RateToUsd
    FROM fin.ExchangeRate
    WHERE Currency = @ccy AND RateDate <= @asOf
    ORDER BY RateDate DESC;
    RETURN @amount * ISNULL(@rate, 1);

CREATE FUNCTION fin.fn_RunningBalance(@accountId INT)
RETURNS TABLE
AS
RETURN (
    SELECT je.EntryDate, jl.Debit, jl.Credit,
           SUM(jl.Debit - jl.Credit) OVER (ORDER BY je.EntryDate, jl.LineId
                ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningBalance
    FROM fin.JournalLine jl
    INNER JOIN fin.JournalEntry je ON je.EntryId = jl.EntryId
    WHERE jl.AccountId = @accountId
);

===OBJECT:fin.usp_PeriodPivot:SQL_STORED_PROCEDURE===
-- dynamic PIVOT via sp_executesql (period columns computed at runtime)
CREATE PROCEDURE fin.usp_PeriodPivot
    @FromDate DATE, @ToDate DATE
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @cols NVARCHAR(MAX), @sql NVARCHAR(MAX);
    SELECT @cols = STRING_AGG(QUOTENAME(pk), N',')
    FROM (SELECT DISTINCT fin.fn_PeriodKey(je.EntryDate) AS pk
          FROM fin.JournalEntry je
          WHERE je.EntryDate BETWEEN @FromDate AND @ToDate) x;
    IF @cols IS NULL SET @cols = N'[none]';

    SET @sql = N'
        SELECT AccountId, ' + @cols + N'
        FROM (
            SELECT jl.AccountId, fin.fn_PeriodKey(je.EntryDate) AS Pk, jl.Debit - jl.Credit AS Amt
            FROM fin.JournalLine jl
            INNER JOIN fin.JournalEntry je ON je.EntryId = jl.EntryId
            WHERE je.EntryDate BETWEEN @f AND @t
        ) s
        PIVOT (SUM(Amt) FOR Pk IN (' + @cols + N')) p
        ORDER BY AccountId';
    EXEC sp_executesql @sql, N'@f DATE, @t DATE', @f = @FromDate, @t = @ToDate;
END;

===OBJECT:fin.usp_PostJournal:SQL_STORED_PROCEDURE===
-- TOP proc: post a balanced journal — sequence, cursor, proc→proc, proc→function, TRY/CATCH
CREATE PROCEDURE fin.usp_PostJournal
    @Memo NVARCHAR(200)
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @entryId INT, @jno BIGINT, @acct INT, @dr DECIMAL(18,2), @cr DECIMAL(18,2);
    BEGIN TRY
        BEGIN TRAN;
        SET @jno = NEXT VALUE FOR fin.JournalSeq;
        INSERT INTO fin.JournalEntry(JournalNo, Memo) VALUES(@jno, @Memo);
        SET @entryId = SCOPE_IDENTITY();

        DECLARE @lines TABLE(AccountId INT, Debit DECIMAL(18,2), Credit DECIMAL(18,2));
        INSERT INTO @lines VALUES (1, 100, 0), (2, 0, 100);

        DECLARE ln_cur CURSOR LOCAL FAST_FORWARD FOR
            SELECT AccountId, Debit, Credit FROM @lines;
        OPEN ln_cur;
        FETCH NEXT FROM ln_cur INTO @acct, @dr, @cr;
        WHILE @@FETCH_STATUS = 0
        BEGIN
            EXEC fin.usp_AddJournalLine @entryId, @acct, @dr, @cr;   -- proc → proc
            FETCH NEXT FROM ln_cur INTO @acct, @dr, @cr;
        END
        CLOSE ln_cur; DEALLOCATE ln_cur;

        -- balance check via correlated aggregate
        IF (SELECT SUM(Debit) - SUM(Credit) FROM fin.JournalLine WHERE EntryId = @entryId) <> 0
            THROW 52001, N'Journal not balanced', 1;

        DECLARE @chk DECIMAL(18,2) = fin.fn_AccountBalance(@acct, SYSDATETIME());  -- proc→function→function
        COMMIT;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK;
        INSERT INTO fin.ReconLog(Action, Amount) VALUES(N'ERROR', NULL);
        THROW;
    END CATCH
END;
===END===
===OBJECT:fin.usp_Reconcile:SQL_STORED_PROCEDURE===
-- MERGE + OUTPUT reconciliation
CREATE PROCEDURE fin.usp_Reconcile
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @audit TABLE(AccountId INT, Act NVARCHAR(10), Amt DECIMAL(18,2));

    MERGE fin.ReconLog AS tgt
    USING (SELECT a.AccountId, SUM(jl.Debit - jl.Credit) AS Bal
           FROM fin.Account a
           LEFT JOIN fin.JournalLine jl ON jl.AccountId = a.AccountId
           GROUP BY a.AccountId) AS src
        ON tgt.AccountId = src.AccountId AND tgt.Action = N'BAL'
    WHEN MATCHED THEN UPDATE SET tgt.Amount = src.Bal, tgt.RunOn = SYSDATETIME()
    WHEN NOT MATCHED BY TARGET THEN
        INSERT(AccountId, Action, Amount) VALUES(src.AccountId, N'BAL', src.Bal)
    OUTPUT inserted.AccountId, N'MRG', inserted.Amount INTO @audit;

    SELECT * FROM @audit;
END;
===END===
===OBJECT:fin.usp_RunningBalanceReport:SQL_STORED_PROCEDURE===
-- proc: running balance report — window funcs, CROSS APPLY, OFFSET/FETCH paging
CREATE PROCEDURE fin.usp_RunningBalanceReport
    @AccountId INT, @Page INT = 1, @PageSize INT = 20
AS
BEGIN
    SET NOCOUNT ON;
    SELECT rb.EntryDate, rb.Debit, rb.Credit, rb.RunningBalance,
           latest.LastAmount,
           LAG(rb.RunningBalance) OVER (ORDER BY rb.EntryDate) AS PrevBalance
    FROM fin.fn_RunningBalance(@AccountId) rb
    CROSS APPLY (SELECT TOP 1 (jl.Debit - jl.Credit) AS LastAmount
                 FROM fin.JournalLine jl WHERE jl.AccountId = @AccountId
                 ORDER BY jl.LineId DESC) latest
    ORDER BY rb.EntryDate
    OFFSET (@Page - 1) * @PageSize ROWS FETCH NEXT @PageSize ROWS ONLY;
END;

CREATE FUNCTION hr.fn_OrgChart(@rootId INT)
RETURNS @tree TABLE(Level INT, EmployeeId INT, ManagerId INT, FullName NVARCHAR(101))
AS
BEGIN
    WITH org AS (
        SELECT 0 AS Lvl, e.EmployeeId, e.ManagerId
        FROM hr.Employee e WHERE e.EmployeeId = @rootId

CREATE PROCEDURE hr.usp_DeptPayroll
    @DeptId INT
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @eid INT, @name NVARCHAR(101), @sal DECIMAL(18,2);
    CREATE TABLE #rpt(EmployeeId INT, FullName NVARCHAR(101), Salary DECIMAL(18,2), Reports INT);

    DECLARE emp_cur CURSOR LOCAL FOR

        FROM hr.Employee e

        FETCH NEXT FROM emp_cur INTO @eid, @sal;
    END
    CLOSE emp_cur;
    DEALLOCATE emp_cur;

    SELECT FullName, SUM(Salary) AS Salary
    FROM #rpt
    GROUP BY GROUPING SETS ((FullName), ());

CREATE PROCEDURE sales.usp_BomReport
    @ProductId INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT e.Level, e.ComponentProductId, comp.Name AS ComponentName, e.Qty,
           SUM(e.Qty) OVER (PARTITION BY e.Level) AS QtyAtLevel,
           ROW_NUMBER() OVER (ORDER BY e.Level, e.ComponentProductId) AS Rn
    FROM sales.fn_ExplodeBom(@ProductId) e
    INNER JOIN sales.Product comp ON comp.ProductId = e.ComponentProductId
    ORDER BY e.Level, e.ComponentProductId;
END;

===OBJECT:sales.usp_FulfillOrder:SQL_STORED_PROCEDURE===-- TOP proc: cursor, proc→proc, proc→function, MERGE, TRY/CATCH, transaction
CREATE PROCEDURE sales.usp_FulfillOrder
    @CustomerId INT
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @orderId INT, @pid INT, @qty INT, @total DECIMAL(18,2), @factor DECIMAL(5,3);
    BEGIN TRY
        BEGIN TRAN;
        EXEC sales.usp_CreateOrder @CustomerId, @orderId OUTPUT;      -- proc → proc

        DECLARE @wanted TABLE(ProductId INT, Qty INT);
        INSERT INTO @wanted VALUES (1,2),(2,1),(3,4);

        DECLARE line_cur CURSOR LOCAL FAST_FORWARD FOR
            SELECT ProductId, Qty FROM @wanted;
        OPEN line_cur;
        FETCH NEXT FROM line_cur INTO @pid, @qty;
        WHILE @@FETCH_STATUS = 0
        BEGIN
            EXEC sales.usp_AddOrderLine @orderId, @pid, @qty;         -- proc → proc (in loop)
            -- keep inventory in sync via MERGE
            MERGE sales.Inventory AS tgt
            USING (SELECT @pid AS ProductId, @qty AS Q) AS src
                ON tgt.ProductId = src.ProductId
            WHEN MATCHED THEN UPDATE SET tgt.QtyOnHand = tgt.QtyOnHand - src.Q
            WHEN NOT MATCHED THEN INSERT(ProductId, QtyOnHand) VALUES(src.ProductId, -src.Q);
            FETCH NEXT FROM line_cur INTO @pid, @qty;
        END
        CLOSE line_cur;
        DEALLOCATE line_cur;

        SET @total  = sales.fn_OrderTotal(@orderId);                  -- proc → function
        SET @factor = sales.fn_DiscountFactor(@total);               -- proc → function (→ function)
        UPDATE sales.SalesOrder
            SET Total = @total * @factor, Status = N'CONFIRMED'
        WHERE OrderId = @orderId;

        UPDATE sales.Customer
            SET TotalSpent = TotalSpent + (@total * @factor),
                Tier = sales.fn_GetCustomerTier(TotalSpent + (@total * @factor))
        WHERE CustomerId = @CustomerId;

        COMMIT;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK;
        INSERT INTO sales.OrderAudit(OrderId, Action, Detail) VALUES(NULL, N'ERROR', ERROR_MESSAGE());
        THROW;
    END CATCH
END;
===END===
===OBJECT:sales.usp_ReorderReport:SQL_STORED_PROCEDURE===-- dynamic SQL via EXEC() (concatenated) + self-join + APPLY + correlated subquery
CREATE PROCEDURE sales.usp_ReorderReport
    @SortCol SYSNAME = N'QtyOnHand'
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @sql NVARCHAR(MAX);
    SET @sql = N'
        SELECT p.ProductId, p.Name, inv.QtyOnHand, inv.ReorderLevel,
               ca.TotalSold,
               (SELECT AVG(p2.UnitPrice) FROM sales.Product p2 WHERE p2.Category = p.Category) AS AvgCatPrice
        FROM sales.Product p
        LEFT JOIN sales.Inventory inv ON inv.ProductId = p.ProductId
        CROSS APPLY (SELECT SUM(ol.Qty) AS TotalSold
                     FROM sales.OrderLine ol WHERE ol.ProductId = p.ProductId) ca
        OUTER APPLY (SELECT TOP 1 s2.Name FROM sales.Product s2
                     WHERE s2.Category = p.Category AND s2.ProductId <> p.ProductId) peer
        WHERE inv.QtyOnHand <= inv.ReorderLevel
        ORDER BY inv.' + QUOTENAME(@SortCol) + N' ASC';
    EXEC (@sql);
END;
===END===
===OBJECT:sales.usp_SearchOrders:SQL_STORED_PROCEDURE===-- dynamic SQL via sp_executesql (prepared execution with parameters) + all join types
CREATE PROCEDURE sales.usp_SearchOrders
    @CustomerName NVARCHAR(100) = NULL, @Status NVARCHAR(20) = NULL, @Region NVARCHAR(40) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @sql NVARCHAR(MAX), @params NVARCHAR(MAX);
    SET @sql = N'
        SELECT o.OrderId, o.OrderNumber, c.Name, c.Region, o.Status, o.Total,
               inv.QtyOnHand
        FROM sales.SalesOrder o
        INNER JOIN sales.Customer c        ON c.CustomerId = o.CustomerId
        LEFT  JOIN sales.OrderLine ol      ON ol.OrderId = o.OrderId
        RIGHT JOIN sales.Product p         ON p.ProductId = ol.ProductId
        FULL  JOIN sales.Inventory inv     ON inv.ProductId = p.ProductId
        CROSS JOIN (SELECT COUNT(*) AS TotalCustomers FROM sales.Customer) cc
        WHERE o.IsDeleted = 0
          AND EXISTS (SELECT 1 FROM sales.OrderLine x WHERE x.OrderId = o.OrderId)';
    IF @CustomerName IS NOT NULL SET @sql += N' AND c.Name LIKE ''%'' + @cn + ''%''';
    IF @Status       IS NOT NULL SET @sql += N' AND o.Status = @st';
    IF @Region       IS NOT NULL SET @sql += N' AND c.Region = @rg';
    SET @sql += N' ORDER BY o.Total DESC';
    SET @params = N'@cn NVARCHAR(100), @st NVARCHAR(20), @rg NVARCHAR(40)';
    EXEC sp_executesql @sql, @params, @cn = @CustomerName, @st = @Status, @rg = @Region;
END;
===END===
===OBJECT:sales.vw_OrderSummary:VIEW===
-- ── Regular view (multi-join + LEFT join) ───────────────────────────────────
CREATE VIEW sales.vw_OrderSummary
AS
    SELECT o.OrderId, o.OrderNumber, c.Name AS CustomerName, c.Region,
           o.Status, o.Total,
           lc.LineCount
    FROM sales.SalesOrder o
    INNER JOIN sales.Customer c ON c.CustomerId = o.CustomerId
    LEFT  JOIN (SELECT OrderId, COUNT(*) AS LineCount
                FROM sales.OrderLine GROUP BY OrderId) lc ON lc.OrderId = o.OrderId
    WHERE o.IsDeleted = 0;
===END===

CREATE VIEW sales.vw_ProductSales
WITH SCHEMABINDING
AS
    SELECT ol.ProductId,
           COUNT_BIG(*)        AS LineCount,
           SUM(ol.Qty)         AS TotalQty,
           SUM(ol.LineTotal)   AS TotalAmount
    FROM sales.OrderLine ol
    GROUP BY ol.ProductId;
